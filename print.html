Uva 100 - The 3n + 1 problem

Statement

Solution

#include <bits/stdc++.h>
#define fastio ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
#define int long long
using namespace std;

int solve(int n,int cnt){
    if(n==1) return cnt;
    else if(n&1) return solve(3*n+1,cnt+1);
    else return solve(n/2,cnt+1);
}
signed main(){
    fastio;
    int l,r;
    while(cin >> l >> r){
        bool check=0;
        if(l>r){
            check=1;
            swap(l,r);
        }
        int ans=0;
        for(int i=l;i<=r;i++){
            int result=solve(i,1);
            ans=max(ans,result);
        }
        if(check) swap(l,r);
        cout << l << " " << r << " " << ans << "\n";
    }
}

    while(n--){
        int l;
        cin >> l;
        vector<int> arr(l);
        for(int &i:arr){
            cin >> i;
        }
        int ans=0;
        for(int i=0;i<l;i++){
            for(int j=i+1;j<l;j++){
                if(arr[i]>arr[j]){
                    swap(arr[i],arr[j]);
                    ans++;
                }
            }
        }
        cout << "Optimal train swapping takes " << ans << " swaps.\n";
    }
}

490 - Rotating Sentences

Statement

Solution

#include <bits/stdc++.h>
#define fastio ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
#define int long long
#define eb emplace_back
using namespace std;

signed main(){
    fastio;
    vector<string> arr;
    vector<int> len;
    string s;
    int maxlen=0;
    while(getline(cin,s)){
        arr.eb(s);
        int cur=s.length();
        len.eb(cur);
        maxlen=max(maxlen,cur);
    }
    int ss=arr.size();
    for(int i=0;i<maxlen;i++){
        for(int j=ss-1;j>=0;j--){
            if(len[j]<=i){
                cout << " ";
            }
            else{
                cout << arr[j][i];
            }
        }
        cout << "\n";
    }
}

948 - Fibonaccimal Base

Statement

Solution

#include <bits/stdc++.h>
#define fastio ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
#define int long long
#define eb emplace_back
using namespace std;

void init(vector<int> &fib){
    fib.eb(0),fib.eb(1);
    int a=0,b=1;
    for(int i=0;i<40;i++){
        fib.eb(a+b);
        swap(a,b);
        b=a+b;
    }
    return;
}
signed main(){
    fastio;
    vector<int> fib;
    init(fib);
    int n,x,s=fib.size();
    cin >> n;
    while(n--){
        cin >> x;
        string ans;
        bool check=0;
        cout << x;
        for(int i=40;i>=2;i--){
            int cur=x/fib[i];
            x%=fib[i];
            if(cur>0){
                ans+=(cur+'0');
                check=1;
            }
            else if(check){
                ans+='0';
            }
        }
        cout << " = " << ans << " (fib)\n";
    }
}

10008 - What's Cryptanalysis?

Statement

Solution

#include <bits/stdc++.h>
#define fastio ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
#define int long long
#define pci pair<char,int>
#define f first
#define s second
#define pb push_back
using namespace std;

bool cmp(pci a,pci b){
    return a.s>b.s;
}
signed main(){
    fastio;
    int n;
    cin >> n;
    cin.ignore();
    map<char,int> mp;
    string s;
    for(int i=0;i<n;i++){
        getline(cin,s);
        int len=s.length();
        transform(s.begin(),s.end(),s.begin(),::toupper);
        for(int j=0;j<len;j++){
            if(s[j]>='A' && s[j]<='Z'){
                if(mp.count(s[j])) mp[s[j]]++;
                else mp[s[j]]=1;
            }
        }
    }
    vector<pci> ans;
    for(auto &v:mp){
        ans.pb({v.f,v.s});
    }
    sort(ans.begin(),ans.end(),cmp);
    for(auto &v:ans){
        cout << v.f << " " << v.s << "\n";
    }
}
